import { Component, ElementRef, ViewChild } from '@angular/core'; import { FormControl, FormGroup, Validators } from '@angular/forms'; @Component({ selector: 'app-error-summary', templateUrl: './error-summary.component.html' }) export class ErrorSummaryComponent { @ViewChild('errorSummary') errorSummary!: ElementRef; submitted = false; form = new FormGroup({ email: new FormControl('', [Validators.required, Validators.email]) }); onSubmit(): void { this.submitted = true; if (this.form.invalid) { setTimeout(() => this.errorSummary.nativeElement.focus(), 50); } } focusField(id: string, e: Event): void { e.preventDefault(); document.getElementById(id)?.focus(); } }

Error Management Best Practices

  • Always shift browser DOM focus to the error summary container (with tabindex="-1") when a form submission fails.
  • Provide anchor links within the summary that jump focus directly to the corresponding input element.
  • Never communicate errors through red text or borders alone; include explicit text instructions.
